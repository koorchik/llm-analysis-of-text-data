## Exactly how to run it (the scratchpad runners are in volatile /tmp — recreate from here)

```bash
cd llm-basic-framework
SUBSET=/tmp/subset-dev-software   # rebuild if wiped:
# npm run make-subset -- --list gold/subsets/dev-software-22.txt \
#   --from ../storage/cert.gov.ua/fetched --to $SUBSET
env INPUT_DIR="$SUBSET" OUTPUT_DIR=../storage/cert.gov.ua/processed/experiments-dev \
  STEPS=streamingNormalizer FLOW=incremental CONDITION=skos-v8-gemma31b-cloud \
  CATEGORIES=Software LLM_PROVIDER=ollama LLM_MODEL=gemma4:31b \
  OLLAMA_CLOUD=1 OLLAMA_NUM_CTX=65536 \
  DECISION_STRATEGY=listwise-graph LISTWISE_PROMPT_ID=listwise-id-v1 \
  CANDIDATE_GENERATOR=union-rr CANDIDATE_K=10 CANDIDATE_MIN_SIM=0 REPAIR=0 \
  SKOS_CATCHUP_EVERY=0 SKOS_CATCHUP_WIDTH=40 REASK_PARENTLESS=1 DECOUPLE=1 \
  EMBEDDINGS=1 EMBEDDINGS_PROVIDER=ollama EMBEDDINGS_MODEL=embeddinggemma \
  DECISIONS_LOG=1 JUDGE_UNRESOLVED=1 ORDER=numeric-id JUDGE_SAMPLES=1 npm start
```
